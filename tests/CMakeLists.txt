cmake_minimum_required(VERSION 3.20)
project(rt_roi_overlay_tests LANGUAGES CXX)
enable_testing()

add_executable(rt_roi_overlay_test rt_roi_overlay_test.cpp)
target_include_directories(rt_roi_overlay_test PRIVATE ..)
target_compile_features(rt_roi_overlay_test PRIVATE cxx_std_20)
add_test(NAME rt_roi_overlay_model COMMAND rt_roi_overlay_test)

find_path(BOOST_INCLUDE_DIR boost/property_tree/ptree.hpp)
if(BOOST_INCLUDE_DIR)
    add_executable(rt_roi_overlay_config_test rt_roi_overlay_config_test.cpp)
    target_include_directories(rt_roi_overlay_config_test PRIVATE .. ${BOOST_INCLUDE_DIR})
    target_compile_features(rt_roi_overlay_config_test PRIVATE cxx_std_20)
    add_test(NAME rt_roi_overlay_config COMMAND rt_roi_overlay_config_test)
else()
    message(STATUS "Boost headers absent: configuration test will not be built")
endif()

find_package(Qt6 QUIET COMPONENTS Widgets Svg)
if(Qt6_FOUND)
    set(OVERLAY_QT_LIBRARIES Qt6::Widgets Qt6::Svg)
else()
    find_package(Qt5 QUIET COMPONENTS Widgets Svg)
    if(Qt5_FOUND)
        set(OVERLAY_QT_LIBRARIES Qt5::Widgets Qt5::Svg)
    endif()
endif()
if(OVERLAY_QT_LIBRARIES)
    add_executable(qt_rt_roi_overlay_test qt_rt_roi_overlay_test.cpp ../preview/qt_rt_roi_overlay.cpp)
    target_include_directories(qt_rt_roi_overlay_test PRIVATE ..)
    target_compile_features(qt_rt_roi_overlay_test PRIVATE cxx_std_20)
    target_link_libraries(qt_rt_roi_overlay_test PRIVATE ${OVERLAY_QT_LIBRARIES})
    add_test(NAME qt_rt_roi_overlay COMMAND qt_rt_roi_overlay_test)
    set_tests_properties(qt_rt_roi_overlay PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
else()
    message(STATUS "Qt Widgets/Svg development files absent: Qt overlay test will not be built")
endif()
